> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Put in place the data plumbing and boot-time wiring that Sprint 1b needs to run the dialogue loop end-to-end. After Sprint 1a: TypeScript domain objects can round-trip through SQLite; the ProviderRegistry auto-populates from keychain at app start; LLM outbound calls succeed under a scoped CSP; a dev README explains how to work on Muse.

**Architecture:** All additions layer on Sprint 0. New surface:

- `src/db/client.ts` — extended with error-invalidation of the memoized `Database` handle
- `src/providers/boot.ts` — reads secrets from keyring, calls `registry.register(new AnthropicProvider(...))` etc. at app start
- `src-tauri/tauri.conf.json` — CSP `connect-src` allowlist for `api.anthropic.com` and `api.deepseek.com`
- `README.md` — dev-facing guide

**Tech Stack:** No new libraries. Continues Tauri 2 + React 19 + TS 5 + vitest + rustls-backed HTTPS in WebView.

```typescript
import type { LibraryTrack, SongOrigin } from "../../types";

export type LibraryTrackRow = {
  id: string;
  path: string;
  origin: SongOrigin;
  title: string | null;
  artist: string | null;
  album: string | null;
  duration_ms: number | null;
  added_at: number;
  metadata_json: string | null;
};

export function toRow(t: LibraryTrack): LibraryTrackRow {
  return {
    id: t.id,
    path: t.path,
    origin: t.origin,
    title: t.title ?? null,
    artist: t.artist ?? null,
    album: t.album ?? null,
    duration_ms: t.duration_ms ?? null,
    added_at: t.added_at,
    metadata_json: t.metadata === undefined ? null : JSON.stringify(t.metadata),
  };
}

export function fromRow(r: LibraryTrackRow): LibraryTrack {
  const t: LibraryTrack = {
    id: r.id,
    path: r.path,
    origin: r.origin,
    added_at: r.added_at,
  };
  if (r.title !== null) t.title = r.title;
  if (r.artist !== null) t.artist = r.artist;
  if (r.album !== null) t.album = r.album;
  if (r.duration_ms !== null) t.duration_ms = r.duration_ms;
  if (r.metadata_json !== null) t.metadata = JSON.parse(r.metadata_json);
  return t;
}
```

```typescript
import type { DialogueTurn } from "../../types";
import { getDb } from "../client";
import { toRow, fromRow, DialogueTurnRow } from "../codec/dialogueTurn";

export async function insertTurn(t: DialogueTurn): Promise<void> {
  const row = toRow(t);
  const db = await getDb();
  await db.execute(
    `INSERT INTO dialogue_turns
     (id, timestamp, user_utterance_json, agent_response_json,
      user_reaction_json, current_emotion_json, emotion_delta_json)
     VALUES (?, ?, ?, ?, ?, ?, ?)`,
    [
      row.id,
      row.timestamp,
      row.user_utterance_json,
      row.agent_response_json,
      row.user_reaction_json,
      row.current_emotion_json,
      row.emotion_delta_json,
    ],
  );
}

export async function getTurn(id: string): Promise<DialogueTurn | null> {
  const db = await getDb();
  const rows = await db.select<DialogueTurnRow>(
    `SELECT id, timestamp, user_utterance_json, agent_response_json,
            user_reaction_json, current_emotion_json, emotion_delta_json
     FROM dialogue_turns WHERE id = ?`,
    [id],
  );
  return rows.length === 0 ? null : fromRow(rows[0]);
}

export async function listRecentTurns(limit: number): Promise<DialogueTurn[]> {
  const db = await getDb();
  const rows = await db.select<DialogueTurnRow>(
    `SELECT id, timestamp, user_utterance_json, agent_response_json,
            user_reaction_json, current_emotion_json, emotion_delta_json
     FROM dialogue_turns ORDER BY timestamp DESC LIMIT ?`,
    [limit],
  );
  return rows.map(fromRow);
}

export async function countTurns(): Promise<number> {
  const db = await getDb();
  const rows = await db.select<{ n: number }>(
    `SELECT COUNT(*) as n FROM dialogue_turns`,
  );
  return rows[0]?.n ?? 0;
}
```

```typescript
export async function upsertSoulState(s: SoulState): Promise<void> {
  const row = toRow(s);
  const db = await getDb();
  await db.execute(
    `INSERT INTO soul_state
       (agent_id, created_at, taste_base_json, dynamic_mood_json, proactive_budget_json, updated_at)
     VALUES (?, ?, ?, ?, ?, ?)
     ON CONFLICT(agent_id) DO UPDATE SET
       taste_base_json = excluded.taste_base_json,
       dynamic_mood_json = excluded.dynamic_mood_json,
       proactive_budget_json = excluded.proactive_budget_json,
       updated_at = excluded.updated_at`,
    [row.agent_id, row.created_at, row.taste_base_json, row.dynamic_mood_json, row.proactive_budget_json, row.updated_at],
  );
}

export async function loadSoulState(agentId: string): Promise<SoulState | null> {
  const db = await getDb();
  const rows = await db.select<SoulStateRow>(
    `SELECT agent_id, created_at, taste_base_json, dynamic_mood_json, proactive_budget_json, updated_at
     FROM soul_state WHERE agent_id = ?`,
    [agentId],
  );
```

```typescript
import type { ProviderId } from "../types";
import { AnthropicProvider } from "./anthropic";
import { DeepSeekProvider } from "./deepseek";
import { ProviderRegistry, registry as defaultRegistry } from "./registry";
import { SECRET_KEYS, getSecret } from "../settings/secrets";

export type BootResult = {
  registered: ProviderId[];
  skipped: { id: ProviderId; reason: string }[];
};

type Options = { target?: ProviderRegistry };

export async function bootProviders(opts: Options = {}): Promise<BootResult> {
  const target = opts.target ?? defaultRegistry;
  const registered: ProviderId[] = [];
  const skipped: { id: ProviderId; reason: string }[] = [];

  const anthropicKey = await getSecret(SECRET_KEYS.anthropicApiKey);
  if (anthropicKey) {
    target.register(new AnthropicProvider({ apiKey: anthropicKey }));
    registered.push("anthropic");
  } else {
    skipped.push({ id: "anthropic", reason: "no api key configured" });
  }

  const deepseekKey = await getSecret(SECRET_KEYS.deepseekApiKey);
  if (deepseekKey) {
    target.register(new DeepSeekProvider({ apiKey: deepseekKey }));
    registered.push("deepseek");
  } else {
    skipped.push({ id: "deepseek", reason: "no api key configured" });
  }

  return { registered, skipped };
}
```

```typescript
/**
 * Stateful in-memory DB shim. We can't run real SQLite in the Vitest jsdom env
 * without a heavier setup, but we CAN run all the string-mangling / codec /
 * repo logic against a Map-backed store that mimics the tables that the SQL
 * INSERT/SELECT statements would populate.
 *
 * This lets us assert: "if I insertTurn(x) then getTurn(x.id), do I get x back?"
 * — end to end through the codec, without SQLite.
 */

type Row = Record<string, unknown>;
const store: Record<string, Row[]> = {
  dialogue_turns: [],
  emotion_snapshots: [],
  soul_state: [],
  library_tracks: [],
};

function matchTable(sql: string): keyof typeof store {
  if (/dialogue_turns/i.test(sql)) return "dialogue_turns";
  if (/emotion_snapshots/i.test(sql)) return "emotion_snapshots";
  if (/soul_state/i.test(sql)) return "soul_state";
  if (/library_tracks/i.test(sql)) return "library_tracks";
  throw new Error(`unknown table in sql: ${sql}`);
}

const fakeDb = {
  execute: vi.fn(async (sql: string, args: unknown[] = []) => {
    const table = matchTable(sql);
    // Very rough SQL parser: pull column names from between the first '(' and ')'
    // in INSERT statements; ON CONFLICT rewrites existing.
    if (/INSERT INTO/i.test(sql)) {
      const colMatch = sql.match(/\(([^)]+)\)\s*VALUES/i);
      if (!colMatch) throw new Error(`can't parse cols: ${sql}`);
      const cols = colMatch[1].split(",").map((s) => s.trim());
      const row: Row = {};
      cols.forEach((c, i) => (row[c] = args[i]));
      if (/ON CONFLICT/i.test(sql) && table === "soul_state") {
        const existingIx = store.soul_state.findIndex((r) => r.agent_id === row.agent_id);
        if (existingIx >= 0) store.soul_state[existingIx] = row;
        else store.soul_state.push(row);
      } else {
        store[table].push(row);
      }
      return { rowsAffected: 1, lastInsertId: 0 };
    }
    throw new Error(`unsupported execute: ${sql}`);
  }),
  select: vi.fn(async <T,>(sql: string, args: unknown[] = []): Promise<T[]> => {
    const table = matchTable(sql);
    if (/COUNT\(\*\)/i.test(sql)) {
      return [{ n: store[table].length } as unknown as T];
    }
    if (/WHERE id = \?/i.test(sql)) {
      return store[table].filter((r) => r.id === args[0]) as T[];
    }
    if (/WHERE agent_id = \?/i.test(sql)) {
      return store[table].filter((r) => r.agent_id === args[0]) as T[];
    }
    if (/WHERE turn_id = \?/i.test(sql)) {
      return store[table].filter((r) => r.turn_id === args[0]) as T[];
    }
    if (/WHERE path = \?/i.test(sql)) {
      return store[table].filter((r) => r.path === args[0]) as T[];
    }
    // no WHERE — return all with LIMIT if present
    let rows = [...store[table]];
    if (/ORDER BY timestamp DESC/i.test(sql)) {
      rows.sort((a, b) => (b.timestamp as number) - (a.timestamp as number));
    }
    if (/ORDER BY added_at ASC/i.test(sql)) {
      rows.sort((a, b) => (a.added_at as number) - (b.added_at as number));
    }
    const limitMatch = sql.match(/LIMIT \?/);
    if (limitMatch) rows = rows.slice(0, args[0] as number);
    return rows as T[];
  }),
};
```
